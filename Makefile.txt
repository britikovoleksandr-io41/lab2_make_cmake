CC := gcc
CFLAGS := -Wall -Wextra -std=c11 -Iinclude
BUILD_DIR := build

LIB_NAME := libcalc.a
APP_NAME := calc_app

LIB := $(BUILD_DIR)/$(LIB_NAME)
APP := $(BUILD_DIR)/$(APP_NAME)

LIB_OBJS := $(BUILD_DIR)/calc.o
APP_OBJS := $(BUILD_DIR)/main.o

.PHONY: all clean run

all: $(LIB) $(APP)

$(BUILD_DIR):
	mkdir -p $(BUILD_DIR)

$(BUILD_DIR)/calc.o: src/calc.c include/calc.h | $(BUILD_DIR)
	$(CC) $(CFLAGS) -c src/calc.c -o $(BUILD_DIR)/calc.o

$(BUILD_DIR)/main.o: src/main.c include/calc.h | $(BUILD_DIR)
	$(CC) $(CFLAGS) -c src/main.c -o $(BUILD_DIR)/main.o

$(LIB): $(LIB_OBJS)
	ar rcs $(LIB) $(LIB_OBJS)

$(APP): $(APP_OBJS) $(LIB)
	$(CC) $(CFLAGS) $(APP_OBJS) $(LIB) -o $(APP)

run: all
	./$(APP)

clean:
	rm -rf $(BUILD_DIR)
